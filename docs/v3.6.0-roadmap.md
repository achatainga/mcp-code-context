# 📋 ROADMAP DETALLADO: mcp-code-context v3.6.0

## 🎯 OBJETIVO PRINCIPAL

Transformar `mcp-code-context` de una herramienta de edición quirúrgica a una **plataforma completa de análisis y edición de código**, cerrando la brecha con `code-index-mcp` mientras se mantiene la ventaja competitiva en edición segura y se corrigen las fallas arquitectónicas críticas identificadas en el análisis adversarial.

---

## 📊 ANÁLISIS DE BRECHA (Gap Analysis)

### Lo que TENEMOS:
- ✅ Edición quirúrgica con Tree-sitter AST
- ✅ Two-phase workflow (seguridad)
- ✅ Backup system (5 versiones)
- ✅ Zero native dependencies (WASM)
- ✅ 5 lenguajes (TS, JS, Python, PHP, Dart)

### Lo que NOS FALTA (vs code-index-mcp):
- ❌ File watcher con auto-refresh
- ❌ Caching persistente
- ❌ Búsqueda avanzada (ripgrep/ugrep)
- ❌ Tests automatizados (Vitest/Jest)
- ❌ Logging estructurado para producción
- ❌ Más lenguajes (Java, Go, Rust, C#)

### PROBLEMAS CRÍTICOS (confirmados con análisis AST):
- 🔴 Estado en memoria (locks, confirmations) no funciona en multi-proceso
- 🔴 Backups en project root causan loops infinitos en hot-reload
- 🔴 Algoritmo LCS en diff causa OOM en archivos grandes (>5000 líneas)
- 🟡 Worker threads para ReDoS son excesivos (30MB RAM cada uno)
- 🟡 Duplicación de código: `walkDir` aparece 3 veces (read.ts, write.ts, compress.ts)
- 🟡 Función monstruo: `renameSymbol` (101 líneas, complejidad ciclomática >15)

### HALLAZGOS DEL ANÁLISIS AST:
**Métricas reales extraídas**:
- `index.ts`: 16 funciones, solo 1 >50 líneas (handleTwoPhaseWrite: 57 líneas)
- `write.ts`: `renameSymbol` = 101 líneas (CRÍTICO), `insertCode` = 53 líneas
- `read.ts`: `searchPattern` = 77 líneas, `readLines` = 60 líneas
- Duplicación: `walkDir` implementado 3 veces (1075-1197 bytes cada uno)

**Veredicto**: NO es un "God Object". Es un orquestador bien estructurado con 1 función problemática.

**Problema**: Workers que viven mucho tiempo filtran memoria.

**Mitigación**: Reciclar workers cada 100 operaciones.

```typescript
// src/utils/fileLock.ts (REESCRIBIR COMPLETO)
import lockfile from 'proper-lockfile';
import { getLogger } from './logger.js';
import path from 'path';
import { tmpdir } from 'os';
import { mkdirSync, existsSync } from 'fs';
import crypto from 'crypto';

const logger = getLogger('file-lock');

export class FileLockManager {
  private lockDir: string;
  private activeLocks: Map<string, () => Promise<void>> = new Map();

  constructor() {
    // Crear directorio de locks en temp del OS
    const projectHash = crypto.createHash('md5')
      .update(process.cwd())
      .digest('hex')
      .substring(0, 8);
    
    this.lockDir = path.join(tmpdir(), `mcp-locks-${projectHash}`);
    
    if (!existsSync(this.lockDir)) {
      mkdirSync(this.lockDir, { recursive: true });
    }

    logger.info({
      msg: 'File lock manager initialized',
      lockDir: this.lockDir
    });
  }

  async acquireLock(filePath: string, timeoutMs: number = 30000): Promise<() => Promise<void>> {
    const normalizedPath = path.resolve(filePath);
    
    logger.debug({
      msg: 'Acquiring lock',
      file: normalizedPath,
      timeout: timeoutMs
    });

    try {
      const release = await lockfile.lock(normalizedPath, {
        stale: timeoutMs,
        retries: {
          retries: 10,
          minTimeout: 100,
          maxTimeout: 1000,
          factor: 2
        },
        lockfilePath: path.join(
          this.lockDir,
          `${crypto.createHash('md5').update(normalizedPath).digest('hex')}.lock`
        )
      });

      this.activeLocks.set(normalizedPath, release);

      logger.info({
        msg: 'Lock acquired',
        file: normalizedPath
      });

      return async () => {
        await release();
        this.activeLocks.delete(normalizedPath);
        logger.debug({
          msg: 'Lock released',
          file: normalizedPath
        });
      };
    } catch (error) {
      logger.error({
        msg: 'Failed to acquire lock',
        file: normalizedPath,
        error: error.message
      });
      throw new Error(`Could not acquire lock for ${filePath}: ${error.message}`);
    }
  }

  async isLocked(filePath: string): Promise<boolean> {
    const normalizedPath = path.resolve(filePath);
    
    try {
      const lockfilePath = path.join(
        this.lockDir,
        `${crypto.createHash('md5').update(normalizedPath).digest('hex')}.lock`
      );
      
      return await lockfile.check(normalizedPath, { lockfilePath });
    } catch {
      return false;
    }
  }

  async releaseAll(): Promise<void> {
    logger.info({
      msg: 'Releasing all locks',
      count: this.activeLocks.size
    });

    const releases = Array.from(this.activeLocks.values());
    await Promise.all(releases.map(release => release()));
    this.activeLocks.clear();
  }
}

// Singleton global
export const fileLockManager = new FileLockManager();
```

// src/utils/fileWalker.ts (NUEVO)
import * as fs from 'node:fs/promises';
import * as path from 'node:path';
import { EXCLUDE_DIRS, SUPPORTED_EXTENSIONS } from './constants.js';
import { SecurityValidator } from '../core/validator.js';
import { getLogger } from './logger.js';

const logger = getLogger('file-walker');

export interface WalkOptions {
  excludeDirs?: string[];
  extensions?: string[];
  maxDepth?: number;
  followSymlinks?: boolean;
}

export async function walkDirectory(
  rootDir: string,
  callback: (filePath: string, stats: fs.Stats) => Promise<void>,
  options: WalkOptions = {}
): Promise<void> {
  const {
    excludeDirs = EXCLUDE_DIRS,
    extensions = SUPPORTED_EXTENSIONS,
    maxDepth = Infinity,
    followSymlinks = false
  } = options;

  async function walk(dir: string, depth: number = 0) {
    if (depth > maxDepth) return;

    const entries = await fs.readdir(dir, { withFileTypes: true });

    for (const entry of entries) {
      const fullPath = path.join(dir, entry.name);

      if (entry.isDirectory()) {
        if (!excludeDirs.includes(entry.name)) {
          await walk(fullPath, depth + 1);
        }
      } else if (entry.isFile() || (followSymlinks && entry.isSymbolicLink())) {
        const ext = path.extname(entry.name);
        if (extensions.includes(ext as any)) {
          const stats = await fs.stat(fullPath);
          await callback(fullPath, stats);
        }
      }
    }
  }

  logger.debug({ msg: 'Starting directory walk', rootDir });
  await walk(rootDir);
  logger.debug({ msg: 'Directory walk completed', rootDir });
}

// src/utils/searchTools.ts (NUEVO)
import which from 'which';
import { exec } from 'child_process';
import { promisify } from 'util';
import { getLogger } from './logger.js';

const execAsync = promisify(exec);
const logger = getLogger('search-tools');

export type SearchTool = 'ripgrep' | 'ugrep' | 'ag' | 'grep' | 'findstr';

let cachedTool: SearchTool | null = null;

export async function detectSearchTool(): Promise<SearchTool> {
  if (cachedTool) return cachedTool;
  
  logger.debug('Detecting available search tools...');
  
  // Orden de preferencia (velocidad + compatibilidad)
  const tools: Array<{ name: SearchTool; command: string }> = [
    { name: 'ripgrep', command: 'rg' },       // Más rápido + cross-platform
    { name: 'ugrep', command: 'ugrep' },      // Muy rápido + cross-platform
    { name: 'ag', command: 'ag' },            // Rápido + cross-platform
  ];
  
  // Fallback específico por plataforma
  if (process.platform === 'win32') {
    tools.push({ name: 'findstr', command: 'findstr' });  // Windows nativo
  } else {
    tools.push({ name: 'grep', command: 'grep' });        // Unix/Linux/macOS nativo
  }
  
  for (const tool of tools) {
    try {
      await which(tool.command);
      logger.info({ msg: 'Search tool detected', tool: tool.name });
      cachedTool = tool.name;
      return tool.name;
    } catch {
      continue;
    }
  }
  
  // Fallback específico por plataforma
  if (process.platform === 'win32') {
    logger.warn('No optimized search tool found, using findstr (Windows native)');
    cachedTool = 'findstr';
    return 'findstr';
  } else {
    logger.warn('No optimized search tool found, using grep (Unix native)');
    cachedTool = 'grep';
    return 'grep';
  }
}

export async function searchWithTool(
  tool: SearchTool,
  pattern: string,
  rootDir: string,
  options: {
    caseInsensitive?: boolean;
    contextLines?: number;
    filePattern?: string;
    maxResults?: number;
  } = {}
): Promise<string> {
  const {
    caseInsensitive = false,
    contextLines = 0,
    filePattern,
    maxResults = 50
  } = options;
  
  let command: string;
  
  switch (tool) {
    case 'ripgrep':
      command = `rg ${caseInsensitive ? '-i' : ''} ${contextLines > 0 ? `-C ${contextLines}` : ''} "${pattern}" "${rootDir}"`;
      if (filePattern) command += ` -g "${filePattern}"`;
      command += ` -m ${maxResults}`;
      break;
      
    case 'ugrep':
      command = `ugrep ${caseInsensitive ? '-i' : ''} ${contextLines > 0 ? `-C ${contextLines}` : ''} "${pattern}" "${rootDir}"`;
      if (filePattern) command += ` --include="${filePattern}"`;
      command += ` -m ${maxResults}`;
      break;
      
    case 'ag':
      command = `ag ${caseInsensitive ? '-i' : ''} ${contextLines > 0 ? `-C ${contextLines}` : ''} "${pattern}" "${rootDir}"`;
      if (filePattern) command += ` --${filePattern}`;
      break;
      
    case 'grep':
      // Unix/Linux/macOS nativo
      command = `grep -r ${caseInsensitive ? '-i' : ''} ${contextLines > 0 ? `-C ${contextLines}` : ''} "${pattern}" "${rootDir}"`;
      if (filePattern) command += ` --include="${filePattern}"`;
      break;
      
    case 'findstr':
      // Windows nativo (limitado - no soporta regex complejas ni contexto)
      command = `findstr /S ${caseInsensitive ? '/I' : ''} "${pattern}" "${rootDir}\\*.*"`;
      if (filePattern) {
        logger.warn('findstr does not support file patterns, results may include unwanted files');
      }
      break;
  }
  
  try {
    const { stdout } = await execAsync(command, { maxBuffer: 10 * 1024 * 1024 });
    return stdout;
  } catch (error: any) {
    // Exit code 1 = no matches (no es error)
    if (error.code === 1) return '';
    throw error;
  }
}

export async function refreshSearchTools(): Promise<SearchTool> {
  cachedTool = null;
  return await detectSearchTool();
}

// src/core/cacheManager.ts
import Database from 'better-sqlite3';
import { tmpdir } from 'os';
import path from 'path';
import crypto from 'crypto';
import { getLogger } from '../utils/logger.js';
import { existsSync, mkdirSync } from 'fs';

const logger = getLogger('cache-manager');

export interface CachedFile {
  path: string;
  mtime: number;
  size: number;
  hash: string;
  symbols: string; // JSON stringified
  updatedAt: number;
}

export class CacheManager {
  private db: Database.Database;
  private cacheDir: string;

  constructor(projectRoot: string) {
    // Hash del project root
    const projectHash = crypto.createHash('md5')
      .update(projectRoot)
      .digest('hex')
      .substring(0, 8);
    
    // Cache en temp del OS
    this.cacheDir = path.join(tmpdir(), 'mcp-cache', projectHash);
    
    if (!existsSync(this.cacheDir)) {
      mkdirSync(this.cacheDir, { recursive: true });
    }

    const dbPath = path.join(this.cacheDir, 'index.db');
    
    logger.info({
      msg: 'Initializing cache',
      dbPath
    });

    this.db = new Database(dbPath);
    this.initSchema();
  }

  private initSchema(): void {
    this.db.exec(`
      CREATE TABLE IF NOT EXISTS files (
        path TEXT PRIMARY KEY,
        mtime INTEGER NOT NULL,
        size INTEGER NOT NULL,
        hash TEXT NOT NULL,
        symbols TEXT,
        updated_at INTEGER NOT NULL
      );
      
      CREATE INDEX IF NOT EXISTS idx_mtime ON files(mtime);
      CREATE INDEX IF NOT EXISTS idx_updated ON files(updated_at);
      CREATE INDEX IF NOT EXISTS idx_hash ON files(hash);
    `);

    logger.debug('Cache schema initialized');
  }

  get(filePath: string): CachedFile | null {
    const stmt = this.db.prepare('SELECT * FROM files WHERE path = ?');
    const result = stmt.get(filePath) as CachedFile | undefined;

    if (result) {
      logger.debug({
        msg: 'Cache hit',
        file: filePath
      });
    } else {
      logger.debug({
        msg: 'Cache miss',
        file: filePath
      });
    }

    return result || null;
  }

  set(file: CachedFile): void {
    const stmt = this.db.prepare(`
      INSERT OR REPLACE INTO files (path, mtime, size, hash, symbols, updated_at)
      VALUES (?, ?, ?, ?, ?, ?)
    `);

    stmt.run(
      file.path,
      file.mtime,
      file.size,
      file.hash,
      file.symbols,
      file.updatedAt
    );

    logger.debug({
      msg: 'Cache updated',
      file: file.path
    });
  }

  invalidate(filePath: string): void {
    const stmt = this.db.prepare('DELETE FROM files WHERE path = ?');
    stmt.run(filePath);

    logger.debug({
      msg: 'Cache invalidated',
      file: filePath
    });
  }

  clear(): void {
    this.db.exec('DELETE FROM files');
    
    logger.info('Cache cleared');
  }

  getStats() {
    const stmt = this.db.prepare('SELECT COUNT(*) as count, SUM(size) as totalSize FROM files');
    const result = stmt.get() as { count: number; totalSize: number };

    return {
      files: result.count,
      totalSize: result.totalSize,
      cacheDir: this.cacheDir
    };
  }

  close(): void {
    this.db.close();
    logger.info('Cache closed');
  }
}

// src/utils/fileWatcher.ts
import chokidar, { FSWatcher } from 'chokidar';
import { getLogger } from './logger.js';
import { EventEmitter } from 'events';

const logger = getLogger('file-watcher');

export interface WatcherConfig {
  enabled: boolean;
  debounceMs: number;
  ignored: string[];
}

export interface FileChangeEvent {
  type: 'add' | 'change' | 'unlink';
  path: string;
  timestamp: number;
}

export class FileWatcherService extends EventEmitter {
  private watcher: FSWatcher | null = null;
  private config: WatcherConfig;
  private debounceTimers: Map<string, NodeJS.Timeout> = new Map();
  private watchedPath: string | null = null;

  constructor(config: Partial<WatcherConfig> = {}) {
    super();
    this.config = {
      enabled: config.enabled ?? true,
      debounceMs: config.debounceMs ?? 500,
      ignored: config.ignored ?? [
        '**/node_modules/**',
        '**/.git/**',
        '**/dist/**',
        '**/build/**',
        '**/.mcp-backups/**',
        '**/*.log'
      ]
    };
  }

  async start(path: string): Promise<void> {
    if (!this.config.enabled) {
      logger.info('File watcher disabled');
      return;
    }

    if (this.watcher) {
      await this.stop();
    }

    logger.info({
      msg: 'Starting file watcher',
      path,
      debounce: this.config.debounceMs
    });

    this.watchedPath = path;

    this.watcher = chokidar.watch(path, {
      ignored: this.config.ignored,
      persistent: true,
      ignoreInitial: true,
      awaitWriteFinish: {
        stabilityThreshold: 300,
        pollInterval: 100
      }
    });

    this.watcher
      .on('add', (filePath) => this.handleChange('add', filePath))
      .on('change', (filePath) => this.handleChange('change', filePath))
      .on('unlink', (filePath) => this.handleChange('unlink', filePath))
      .on('error', (error) => {
        logger.error({
          msg: 'File watcher error',
          error: error.message
        });
      });

    await new Promise((resolve) => {
      this.watcher!.on('ready', resolve);
    });

    logger.info('File watcher ready');
  }

  private handleChange(type: FileChangeEvent['type'], filePath: string): void {
    // Debounce para evitar múltiples eventos del mismo archivo
    const existingTimer = this.debounceTimers.get(filePath);
    if (existingTimer) {
      clearTimeout(existingTimer);
    }

    const timer = setTimeout(() => {
      logger.debug({
        msg: 'File change detected',
        type,
        file: filePath
      });

      this.emit('change', {
        type,
        path: filePath,
        timestamp: Date.now()
      } as FileChangeEvent);

      this.debounceTimers.delete(filePath);
    }, this.config.debounceMs);

    this.debounceTimers.set(filePath, timer);
  }

  async stop(): Promise<void> {
    if (!this.watcher) return;

    logger.info('Stopping file watcher');

    for (const timer of this.debounceTimers.values()) {
      clearTimeout(timer);
    }
    this.debounceTimers.clear();

    await this.watcher.close();
    this.watcher = null;
    this.watchedPath = null;

    logger.info('File watcher stopped');
  }

  getStatus() {
    return {
      enabled: this.config.enabled,
      active: this.watcher !== null,
      watchedPath: this.watchedPath,
      debounceMs: this.config.debounceMs
    };
  }

  updateConfig(config: Partial<WatcherConfig>): void {
    this.config = { ...this.config, ...config };
    logger.info({
      msg: 'File watcher config updated',
      config: this.config
    });
  }
}

// src/utils/safeRegex.ts (MEJORAR)
import { Worker } from 'worker_threads';
import { getLogger } from './logger.js';

const logger = getLogger('safe-regex');

class WorkerPool {
  private workers: Worker[] = [];
  private available: Worker[] = [];
  private maxWorkers = 4; // Límite razonable

  constructor() {
    // Pre-crear workers
    for (let i = 0; i < this.maxWorkers; i++) {
      const worker = new Worker('./safeRegex.worker.js');
      this.workers.push(worker);
      this.available.push(worker);
    }

    logger.info({
      msg: 'Worker pool initialized',
      size: this.maxWorkers
    });
  }

  async execute(pattern: string, text: string, timeout: number): Promise<boolean> {
    if (this.available.length === 0) {
      logger.warn('No workers available, waiting...');
      await new Promise(resolve => setTimeout(resolve, 100));
      return this.execute(pattern, text, timeout);
    }

    const worker = this.available.pop()!;

    try {
      const result = await this.runInWorker(worker, pattern, text, timeout);
      return result;
    } finally {
      this.available.push(worker);
    }
  }

  private runInWorker(worker: Worker, pattern: string, text: string, timeout: number): Promise<boolean> {
    return new Promise((resolve, reject) => {
      const timer = setTimeout(() => {
        reject(new Error('ReDoS timeout'));
      }, timeout);

      worker.once('message', (result) => {
        clearTimeout(timer);
        resolve(result);
      });

      worker.once('error', (error) => {
        clearTimeout(timer);
        reject(error);
      });

      worker.postMessage({ pattern, text });
    });
  }

  destroy(): void {
    for (const worker of this.workers) {
      worker.terminate();
    }
    this.workers = [];
    this.available = [];
  }
}

// Singleton
const workerPool = new WorkerPool();

  return workerPool.execute(pattern, text, timeout);
}

import { getLogger } from './logger.js';
import * as fs from 'node:fs/promises';
import { walkDirectory } from './fileWalker.js';

const logger = getLogger('fuzzy-search');

export interface FuzzySearchOptions {
  threshold?: number;      // 0.0 (exact) - 1.0 (anything), default: 0.6
  maxResults?: number;     // default: 50
  includeScore?: boolean;  // default: false
}

export interface FuzzyMatch {
  filePath: string;
  line: number;
  content: string;
  score?: number;  // 0.0 (perfect) - 1.0 (poor)
}

export async function fuzzySearchCode(
  rootDir: string,
  query: string,
  options: FuzzySearchOptions = {}
): Promise<FuzzyMatch[]> {
  const {
    threshold = 0.6,
    maxResults = 50,
    includeScore = false
  } = options;
  
  logger.debug({
    msg: 'Starting fuzzy search',
    query,
    threshold,
    rootDir
  });
  
  // Recolectar todas las líneas de código
  const codeLines: Array<{ filePath: string; line: number; content: string }> = [];
  
  await walkDirectory(rootDir, async (filePath) => {
    const content = await fs.readFile(filePath, 'utf-8');
    const lines = content.split('\n');
    
    lines.forEach((line, index) => {
      if (line.trim().length > 0) {
        codeLines.push({
          filePath,
          line: index + 1,
          content: line
        });
      }
    });
  });
  
  logger.debug({
    msg: 'Code lines collected',
    totalLines: codeLines.length
  });
  
  // Configurar Fuse.js
  const fuse = new Fuse(codeLines, {
    keys: ['content'],
    threshold,
    includeScore,
    ignoreLocation: true,  // Buscar en cualquier parte de la línea
    minMatchCharLength: 2
  });
  
  // Buscar
  const results = fuse.search(query, { limit: maxResults });
  
  logger.info({
    msg: 'Fuzzy search completed',
    query,
    matches: results.length
  });
  
  // Formatear resultados
  return results.map(result => ({
    filePath: result.item.filePath,
    line: result.item.line,
    content: result.item.content,
    ...(includeScore && { score: result.score })
  }));
}

// src/index.ts - handleTwoPhaseWrite
async function handleTwoPhaseWrite(
  args: Record<string, unknown>,
  operationName: string,
  executeWrite: () => Promise<any>
) {
  const confirm = Boolean(args.confirm);
  const token = args.confirmationToken ? String(args.confirmationToken) : undefined;

  if (confirm) {
    if (!token) throw new Error("confirmationToken is required when confirm=true");
    const pendingOp = globalConfirmationStore.consumePending(token);
    if (!pendingOp) throw new Error(`Invalid or expired confirmation token: ${token}`);
    
    if (pendingOp.pendingWrites && pendingOp.pendingWrites.length > 0) {
      for (const pw of pendingOp.pendingWrites) {
        await BackupManager.createBackup(pw.filePath, String(args.projectRoot));
        await writeFile(pw.filePath, pw.newContent);
      }
    } else {
      await BackupManager.createBackup(pendingOp.filePath, String(args.projectRoot));
      await writeFile(pendingOp.filePath, pendingOp.newContent);
    }

    const fileCount = pendingOp.pendingWrites?.length || 1;
    
    // CAMBIO: No retornar diff en Phase 2
    return {
      content: [{ 
        type: "text", 
        text: `✅ Success. Changes applied to ${fileCount} file(s).` 
        // SIN DIFF - Ya se mostró en Phase 1
      }],
    };
  }

  // Phase 1: Dry-run (sin cambios)
  const result = await executeWrite();
  if (!result.success) throw new Error(result.error);

  const hadConflict = globalConfirmationStore.hasConflictingPending(String(args.filePath));

  const newToken = globalConfirmationStore.storePending({
    filePath: String(args.filePath),
    operation: operationName,
    symbolName: args.symbolName ? String(args.symbolName) : undefined,
    newContent: result.newContent!,
    diff: result.diff!,
    pendingWrites: result.pendingWrites,
  });

  const conflictWarning = hadConflict
    ? `\n⚠️  A previous pending token for this file was invalidated. Only this new token is valid.`
    : "";

  return {
    content: [{ 
      type: "text", 
      text: `DRY RUN SUCCESSFUL.${conflictWarning}\nTo apply these changes, call this tool again with confirm=true and confirmationToken="${newToken}"\n\nDiff:\n${result.diff}`

  const diffs = dmp.diff_main(oldText, newText);
  dmp.diff_cleanupSemantic(diffs);

  const patch = dmp.patch_make(oldText, diffs);
  const patchText = dmp.patch_toText(patch);

  logger.info({
    msg: 'Diff generated',
    patches: patch.length,
    size: patchText.length
  });

  return patchText;
}

// NUEVA: Diff compacto (solo cambios)
export function generateCompactDiff(oldText: string, newText: string): string {
  logger.debug({
    msg: 'Generating compact diff',
    oldLength: oldText.length,
    newLength: newText.length
  });

  const diffs = dmp.diff_main(oldText, newText);
  dmp.diff_cleanupSemantic(diffs);

  // Solo mostrar líneas cambiadas (sin contexto)
  const compactLines: string[] = [];
  
  for (const [op, text] of diffs) {
    if (op === 0) continue; // Ignorar líneas sin cambios
    
    const prefix = op === 1 ? '+ ' : '- ';
    const lines = text.split('\n').filter(line => line.trim().length > 0);
    
    for (const line of lines) {
      compactLines.push(prefix + line);
    }
  }

  const result = compactLines.join('\n');
  
  logger.info({
    msg: 'Compact diff generated',
    originalSize: oldText.length + newText.length,
    compactSize: result.length,
    reduction: `${Math.round((1 - result.length / (oldText.length + newText.length)) * 100)}%`
  });

  return result;
}

// NUEVA: Diff inteligente (compacto si es grande)
export function generateSmartDiff(oldText: string, newText: string): string {
  const fullDiff = generateDiff(oldText, newText);
  
  // Si el diff es >2KB, usar versión compacta
  if (fullDiff.length > 2048) {
    logger.info('Using compact diff (full diff too large)');
    return generateCompactDiff(oldText, newText);
  }
  
  return fullDiff;
}

export async function compressRepository(params: {
  directoryPath: string;
  registry: ParserRegistry;
  format: 'xml' | 'markdown';
  maxDepth?: number;
  autoOptimize?: boolean; // NUEVO
}): Promise<CompressResult> {
  const { directoryPath, registry, format, maxDepth = 10, autoOptimize = true } = params;
  
  // Primera pasada: contar archivos y estimar tamaño
  let fileCount = 0;
  let estimatedSize = 0;
  
  await walkDirectory(directoryPath, async (filePath) => {
    fileCount++;
    const stats = await fs.stat(filePath);
    estimatedSize += stats.size;
  }, { maxDepth });
  
  logger.info({
    msg: 'Repository scan',
    fileCount,
    estimatedSize,
    autoOptimize
  });
  
  // Auto-optimizar si es muy grande
  let includeSymbols = true;
  
  if (autoOptimize) {
    if (fileCount > 100 || estimatedSize > 1024 * 1024) {
      includeSymbols = false;
      logger.info('Auto-optimization: Disabling symbols (large repository)');
    }
  }
  
  // Generar output
  const output = await generateOutput({
    directoryPath,
    registry,
    format,
    maxDepth,
    includeSymbols
  });
  
  // Si aún es muy grande, truncar
  if (output.length > 100 * 1024) {
    logger.warn('Output too large, truncating...');
    return {
      success: true,
      content: output.substring(0, 100 * 1024) + '\n\n--- Output truncated (too large) ---'
    };
  }
  
  return {
    success: true,
    content: output
  };
}

// src/utils/searchTools.ts (MEJORAR)
export async function detectSearchTool(): Promise<SearchTool> {
  if (cachedTool) return cachedTool;
  
  logger.debug('Detecting available search tools...');
  
  const isWindows = process.platform === 'win32';
  
  // Orden de preferencia (velocidad + compatibilidad cross-platform)
  const tools: Array<{ name: SearchTool; command: string }> = [
    { name: 'ripgrep', command: 'rg' },       // Más rápido + cross-platform
    { name: 'ugrep', command: 'ugrep' },      // Muy rápido + cross-platform
    { name: 'ag', command: 'ag' },            // Rápido + cross-platform
  ];
  
  // Fallback específico por plataforma
  if (isWindows) {
    tools.push({ name: 'findstr', command: 'findstr' });  // Windows nativo
  } else {
    tools.push({ name: 'grep', command: 'grep' });        // Unix/Linux/macOS nativo
  }
  
  for (const tool of tools) {
    try {
      await which(tool.command);
      logger.info({ msg: 'Search tool detected', tool: tool.name, platform: process.platform });
      cachedTool = tool.name;
      return tool.name;
    } catch {
      continue;
    }
  }
  
  // Fallback final
  if (isWindows) {
    logger.warn('No optimized search tool found, using findstr (Windows native)');
    cachedTool = 'findstr';
    return 'findstr';
  } else {
    logger.warn('No optimized search tool found, using grep (Unix native)');
    cachedTool = 'grep';
    return 'grep';
  }
}

export async function searchWithTool(
  tool: SearchTool,
  pattern: string,
  rootDir: string,
  options: SearchOptions = {}
): Promise<string> {
  const { caseInsensitive = false, contextLines = 0, filePattern, maxResults = 50 } = options;
  
  let command: string;
  
  switch (tool) {
    case 'ugrep':
    case 'ripgrep':
    case 'ag':
      // ... código existente ...
      break;
      
    case 'grep':
      // Unix/Linux/macOS (nativo)
      command = `grep -r ${caseInsensitive ? '-i' : ''} ${contextLines > 0 ? `-C ${contextLines}` : ''} "${pattern}" "${rootDir}"`;
      if (filePattern) command += ` --include="${filePattern}"`;
      break;
      
    case 'findstr':
      // Windows (nativo, pero limitado)
      // NOTA: findstr no soporta regex complejas ni contexto
      command = `findstr /S ${caseInsensitive ? '/I' : ''} "${pattern}" "${rootDir}\\*.*"`;
      if (filePattern) {
        // findstr no soporta --include, filtrar después
        logger.warn('findstr does not support file patterns, results may include unwanted files');
      }
      break;
  }
  
  try {
    const { stdout } = await execAsync(command, { maxBuffer: 10 * 1024 * 1024 });
    return stdout;
  } catch (error: any) {
    if (error.code === 1) return ''; // No matches
    throw error;
  }
}
